"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str], k: int | None = None) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    seen: dict[str, str] = {}
    for name in names:
        key = name.lower()
        kept = seen.get(key)
        if kept is None:
            seen[key] = name
        elif name.isupper() and not kept.isupper():
            seen[key] = name

    result = sorted(seen.values(), key=lambda s: (len(s), s.lower()))
    if k is not None:
        return result[:k]
    return result
