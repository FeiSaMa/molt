"""Distinct count helpers."""

from __future__ import annotations


def distinct_count(names: list[str]) -> int:
    """Return the number of distinct names under this package's grouping.

    Names are grouped case-insensitively: every spelling of a name counts
    as a single entry, so only the number of case-insensitive groups is
    reported.

    >>> distinct_count(["cc", "bb"])
    2
    >>> distinct_count(["dd", "dd"])
    1
    """
    return len({name.lower() for name in names})
