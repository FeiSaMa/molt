"""Merge and dedupe helpers."""

from __future__ import annotations


def dedupe_merge(a: list[str], b: list[str]) -> list[str]:
    """Merge and dedupe two name lists in this package's preferred sequence.

    >>> dedupe_merge(["cc"], ["bb"])
    ['bb', 'cc']
    >>> dedupe_merge(["c"], ["b"])
    ['b', 'c']
    """
    # 1-2. Group case-insensitively; within each group keep the writing
    # with the smallest character encoding.
    groups: dict[str, str] = {}
    for name in a + b:
        key = name.casefold()
        if key not in groups or name < groups[key]:
            groups[key] = name
    # 3-4. Sort by written length, then by case-insensitive code points.
    result = list(groups.values())
    result.sort(key=lambda s: (len(s), s.casefold()))
    return result
