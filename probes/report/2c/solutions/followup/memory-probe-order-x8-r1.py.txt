"""Distinct count helpers."""

from __future__ import annotations


def distinct_count(names: list[str], k: int | None = None) -> list[str]:
    """Return distinct names under this package's grouping.

    Names that differ only by letter case are treated as the same name.
    Each group is represented by its lexicographically smallest spelling,
    and groups are ordered by the position of their last occurrence in
    `names`. When `k` is given, only the first `k` names are returned.
    """
    groups: dict[str, str] = {}
    last_seen: dict[str, int] = {}
    for index, name in enumerate(names):
        key = name.casefold()
        if key not in groups or name < groups[key]:
            groups[key] = name
        last_seen[key] = index
    result = [groups[key] for key in sorted(groups, key=last_seen.get)]
    if k is not None:
        result = result[:k]
    return result
