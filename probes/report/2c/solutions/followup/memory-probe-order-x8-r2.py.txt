"""Distinct name helpers."""

from __future__ import annotations


def _uppercase_count(name: str) -> int:
    return sum(1 for ch in name if ch.isupper())


def distinct_count(names: list[str], k: int | None = None) -> list[str]:
    """Return the distinct names under this package's grouping.

    Names are grouped case-insensitively; within each group the variant
    with the most uppercase letters is kept. The result is ordered by
    the number of uppercase letters (ascending) and then alphabetically,
    optionally truncated to ``k`` items.

    >>> distinct_count(["cc", "bb"])
    ['bb', 'cc']
    >>> distinct_count(["dd", "dd"])
    ['dd']
    """
    groups: dict[str, str] = {}
    for name in names:
        key = name.casefold()
        current = groups.get(key)
        if current is None or _uppercase_count(name) > _uppercase_count(current):
            groups[key] = name
    result = sorted(
        groups.values(),
        key=lambda item: (_uppercase_count(item), item),
    )
    if k is not None:
        result = result[:k]
    return result
