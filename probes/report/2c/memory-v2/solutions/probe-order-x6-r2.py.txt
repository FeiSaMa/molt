"""Merge and dedupe helpers."""

from __future__ import annotations


def dedupe_merge(a: list[str], b: list[str], k: int | None = None) -> list[str]:
    """Merge and dedupe two name lists in this package's preferred sequence.

    * Case-insensitive dedupe; keep the all-uppercase variant when present.
    * Sort by length ascending, then by lowercase lexicographic order.
    * ``k`` is only an upper bound on the number of returned items.

    >>> dedupe_merge(["cc"], ["bb"])
    ['bb', 'cc']
    >>> dedupe_merge(["c"], ["b"])
    ['b', 'c']
    """
    picked: dict[str, str] = {}
    for item in list(a) + list(b):
        key = item.lower()
        if key not in picked:
            picked[key] = item
        elif item.isupper() and not picked[key].isupper():
            picked[key] = item
    ordered = sorted(picked.values(), key=lambda s: (len(s), s.lower()))
    return ordered[:k] if k is not None else ordered
