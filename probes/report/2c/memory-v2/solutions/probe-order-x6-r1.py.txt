"""Merge and dedupe helpers."""

from __future__ import annotations


def dedupe_merge(a: list[str], b: list[str], k: int | None = None) -> list[str]:
    """Merge and dedupe two name lists in this package's preferred sequence.

    >>> dedupe_merge(["cc"], ["bb"])
    ['bb', 'cc']
    >>> dedupe_merge(["c"], ["b"])
    ['b', 'c']
    """
    seen: dict[str, str] = {}
    for word in a + b:
        key = word.lower()
        if key not in seen or (word.isupper() and not seen[key].isupper()):
            seen[key] = word
    items = sorted(seen.values(), key=lambda w: (len(w), w.lower()))
    if k is not None:
        items = items[:k]
    return items
