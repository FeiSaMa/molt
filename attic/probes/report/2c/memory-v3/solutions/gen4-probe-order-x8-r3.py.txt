"""Distinct count helpers."""

from __future__ import annotations


def distinct_count(names: list[str], keep: int | None = None) -> int:
    """Return the number of distinct names under this package's grouping.

    Names are grouped case-insensitively, keeping for each group the
    spelling with the smallest character encoding. Groups are ordered by
    written length, ties by case-insensitive encoding, and truncated to
    ``keep`` items (after grouping and sorting) when given.

    >>> distinct_count(["cc", "bb"])
    2
    >>> distinct_count(["dd", "dd"])
    1
    """
    groups: dict[str, str] = {}
    for name in names:
        key = name.casefold()
        if key not in groups or name < groups[key]:
            groups[key] = name

    results = sorted(groups.values(), key=lambda name: (len(name), name.casefold()))
    if keep is not None:
        results = results[:keep]
    return len(results)
