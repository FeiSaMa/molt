"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str], limit: int | None = None) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    groups: dict[str, str] = {}
    for name in names:
        key = name.casefold()
        if key not in groups or name < groups[key]:
            groups[key] = name
    result = sorted(groups.values(), key=lambda name: (len(name), name.casefold()))
    if limit is not None:
        result = result[:limit]
    return result
