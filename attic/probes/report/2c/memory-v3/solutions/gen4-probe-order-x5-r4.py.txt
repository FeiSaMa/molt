"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str]) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    groups: dict[str, str] = {}
    for name in names:
        key = name.lower()
        if key not in groups or name < groups[key]:
            groups[key] = name
    return sorted(groups.values(), key=lambda s: (len(s), s.lower()))
