"""Extrema and dedupe helpers."""

from __future__ import annotations


def dedupe_min(names: list[str]) -> str | None:
    """Return the smallest distinct name in this package's preferred sequence, or None when empty.

    >>> dedupe_min(["cc", "bb"])
    'bb'
    >>> dedupe_min([]) is None
    True
    """
    if not names:
        return None
    groups: dict[str, str] = {}
    for name in names:
        key = name.casefold()
        current = groups.get(key)
        if current is None or name < current:
            groups[key] = name
    sequence = sorted(groups.values(), key=lambda n: (len(n), n.casefold()))
    return sequence[0]
