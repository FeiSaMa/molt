"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str], k: int | None = None) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    preferred: dict[str, str] = {}
    for name in names:
        key = name.lower()
        if key not in preferred:
            preferred[key] = name
        elif name.isupper():
            preferred[key] = name

    result = sorted(preferred.values(), key=lambda name: (len(name), name.lower()))
    if k is not None:
        result = result[:k]
    return result
