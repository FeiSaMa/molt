"""Extrema and dedupe helpers."""

from __future__ import annotations


def dedupe_min(names: list[str], k: int | None = None) -> str | list[str] | None:
    """Return the smallest distinct names in this package's preferred sequence.

    Names are deduplicated case-insensitively; when several casings of the
    same word occur, an ALL-CAPS variant is preferred.  Survivors are ordered
    by length (ascending), ties broken by their lowercased form (lexicographic
    ascending).  ``k`` caps the number of returned names and never pads:
    fewer unique names are returned in full.  Without ``k`` the single
    smallest name is returned, or None when the input is empty.

    >>> dedupe_min(["cc", "bb"])
    'bb'
    >>> dedupe_min(["Apple", "apple", "APPLE", "banana"], 2)
    ['APPLE', 'banana']
    >>> dedupe_min([]) is None
    True
    """
    if not names:
        return None if k is None else []
    seen: dict[str, str] = {}
    for name in names:
        key = name.lower()
        if key not in seen:
            seen[key] = name
        elif name.isupper() and not seen[key].isupper():
            seen[key] = name
    ordered = sorted(seen.values(), key=lambda s: (len(s), s.lower()))
    return ordered[0] if k is None else ordered[:k]
