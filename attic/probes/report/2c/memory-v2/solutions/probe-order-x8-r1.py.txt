"""Distinct name helpers."""

from __future__ import annotations


def distinct_count(names: list[str], k: int | None = None) -> list[str]:
    """Return the distinct names, deduplicated case-insensitively.

    Within each case-insensitive group the all-uppercase variant is kept
    when present.  Results are sorted by length ascending (ties broken by
    the lowercase form) and truncated to at most ``k`` items when ``k`` is
    given.  Fewer than ``k`` distinct names are returned as-is, with no
    padding or repeats.

    >>> distinct_count(["cc", "bb"])
    ['bb', 'cc']
    >>> distinct_count(["dd", "dd"])
    ['dd']
    >>> distinct_count(["Banana", "apple", "banana", "APPLE"], 2)
    ['APPLE', 'Banana']
    """
    groups: dict[str, str] = {}
    for name in names:
        key = name.lower()
        current = groups.get(key)
        if current is None or (name.isupper() and not current.isupper()):
            groups[key] = name

    unique = list(groups.values())
    unique.sort(key=lambda item: (len(item), item.lower()))
    if k is not None:
        unique = unique[:k]
    return unique
