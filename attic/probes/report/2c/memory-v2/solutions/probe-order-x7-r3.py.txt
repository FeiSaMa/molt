"""Extrema and dedupe helpers."""

from __future__ import annotations


def dedupe_min(names: list[str], k: int | None = None) -> list[str] | str | None:
    """Return distinct names in this package's preferred sequence.

    Duplicates are matched case-insensitively; when a group has an all-caps
    variant, that variant is kept.  Results are sorted by length and then by
    their lowercase form.

    Without k the single smallest name is returned (None when empty); with k
    a list of at most k names is returned.

    >>> dedupe_min(["cc", "bb"])
    'bb'
    >>> dedupe_min([]) is None
    True
    """
    chosen: dict[str, str] = {}
    for name in names:
        key = name.lower()
        current = chosen.get(key)
        if current is None or (name == name.upper() and current != current.upper()):
            chosen[key] = name
    ordered = sorted(chosen.values(), key=lambda s: (len(s), s.lower()))
    if k is None:
        return ordered[0] if ordered else None
    return ordered[:k]
