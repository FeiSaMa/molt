"""Extrema and dedupe helpers."""

from __future__ import annotations


def dedupe_min(names: list[str]) -> str | None:
    """Return the smallest distinct name in this package's preferred sequence, or None when empty.

    >>> dedupe_min(["cc", "bb"])
    'bb'
    >>> dedupe_min([]) is None
    True
    """
    if not names:
        return None
    unique: dict[str, str] = {}
    for name in names:
        key = name.lower()
        if key not in unique or name.isupper():
            unique[key] = name
    return min(unique.values(), key=lambda s: (len(s), s.lower()))
