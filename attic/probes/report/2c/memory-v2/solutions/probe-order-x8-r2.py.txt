"""Distinct count helpers."""

from __future__ import annotations


def distinct_count(names: list[str], k: int) -> list[str]:
    """Return up to k distinct names under this package's grouping.

    Grouping is case-insensitive; when a group contains an all-uppercase
    variant, that variant is kept. The result is sorted by length, then by
    the lowercase form lexicographically.

    >>> distinct_count(["cc", "bb"], 2)
    ['bb', 'cc']
    >>> distinct_count(["dd", "dd"], 1)
    ['dd']
    >>> distinct_count(["apple", "APPLE", "banana"], 2)
    ['APPLE', 'banana']
    """
    seen: dict[str, str] = {}
    for name in names:
        key = name.lower()
        if key not in seen:
            seen[key] = name
        elif name.isupper() and not seen[key].isupper():
            seen[key] = name
    uniq = sorted(seen.values(), key=lambda s: (len(s), s.lower()))
    return uniq[:k]
