"""Distinct count helpers."""

from __future__ import annotations

from collections import Counter


def distinct_count(names: list[str], k: int | None = None) -> list[str]:
    """Return distinct names under this package's grouping.

    Names that differ only by case form one group. Each group is
    represented by its ASCII-smallest spelling (uppercase preferred).
    Groups are ordered by frequency ascending (rarest first); ties are
    broken by the representative spelling ascending. When ``k`` is
    given, only the first ``k`` representatives are returned.

    >>> distinct_count(["cc", "bb"])
    ['bb', 'cc']
    >>> distinct_count(["dd", "dd"])
    ['dd']
    """
    counts = Counter(name.casefold() for name in names)
    spellings: dict[str, set[str]] = {}
    for name in names:
        spellings.setdefault(name.casefold(), set()).add(name)
    items = sorted(
        ((counts[key], min(variants)) for key, variants in spellings.items()),
        key=lambda item: (item[0], item[1]),
    )
    result = [rep for _, rep in items]
    if k is not None:
        result = result[:k]
    return result
