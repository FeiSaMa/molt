"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str], k: int = 1) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    seen: dict[str, str] = {}
    for name in names:
        key = name[:k].lower()
        if key not in seen or name.isupper():
            seen[key] = name
    return sorted(seen.values(), key=lambda s: (s.lower() != s, s.lower()))
