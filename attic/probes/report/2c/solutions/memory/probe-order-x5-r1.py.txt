"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str], k: int | None = None) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    ordered = sorted(names, key=lambda s: (len(s), s.casefold(), s))
    result: list[str] = []
    seen: set[str] = set()
    for name in ordered:
        key = name.casefold()
        if key not in seen:
            seen.add(key)
            result.append(name)
    return result if k is None else result[:k]
