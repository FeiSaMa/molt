"""Merge and dedupe helpers."""

from __future__ import annotations


def dedupe_merge(a: list[str], b: list[str]) -> list[str]:
    """Merge and dedupe two name lists in this package's preferred sequence.

    >>> dedupe_merge(["cc"], ["bb"])
    ['bb', 'cc']
    >>> dedupe_merge(["c"], ["b"])
    ['b', 'c']
    """
    best: dict[str, str] = {}
    for name in a + b:
        key = name.casefold()
        if key not in best:
            best[key] = name
        elif name == name.upper() and best[key] != best[key].upper():
            best[key] = name
    return sorted(best.values(), key=lambda s: (s != s.lower(), s.casefold()))
