"""Distinct count helpers."""

from __future__ import annotations


def distinct_count(names: list[str], k: int | None = None) -> list[str]:
    """Return the distinct names under this package's grouping.

    Names are grouped case-insensitively: ``Alpha``, ``ALPHA`` and
    ``alpha`` are the same name.  Each group is represented by its
    all-uppercase member when one exists, otherwise by its first member.
    Groups are listed in reverse order of first appearance; if *k* is
    given, only the first *k* groups are returned.

    >>> distinct_count(["cc", "bb"])
    ['bb', 'cc']
    >>> distinct_count(["dd", "dd"])
    ['dd']
    """
    groups: dict[str, list[str]] = {}
    for name in names:
        groups.setdefault(name.casefold(), []).append(name)

    result: list[str] = []
    for members in groups.values():
        upper = next((member for member in members if member.isupper()), None)
        result.append(upper if upper is not None else members[0])

    result.reverse()
    if k is not None:
        return result[:k]
    return result
