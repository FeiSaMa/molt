"""Unique helpers."""

from __future__ import annotations


def unique(names: list[str], k: int | None = None) -> list[str]:
    """Drop duplicate names and return them in this package's preferred sequence.

    Names are compared case-insensitively; of several case variants the one
    with the most uppercase letters is kept.  The survivors are ordered by
    how often their group occurred in the input (least frequent first) and
    then case-insensitively alphabetically.  ``k`` limits the result length.

    >>> unique(["cc", "bb"])
    ['bb', 'cc']
    >>> unique(["dd", "ee"])
    ['dd', 'ee']
    """
    groups: dict[str, list] = {}
    for name in names:
        key = name.casefold()
        if key not in groups:
            groups[key] = [name, 0]
        groups[key][1] += 1
        current = groups[key][0]
        if sum(ch.isupper() for ch in name) > sum(ch.isupper() for ch in current):
            groups[key][0] = name

    result = [
        rep
        for rep, _count in sorted(
            groups.values(), key=lambda item: (item[1], item[0].casefold())
        )
    ]
    if k is not None:
        result = result[:k]
    return result
