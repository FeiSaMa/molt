"""Extrema and dedupe helpers."""

from __future__ import annotations


def dedupe_min(names: list[str], k: int | None = None) -> list[str] | None:
    """Return distinct names in this package's preferred sequence, or None when empty.

    Names are deduplicated case-insensitively, keeping the canonical
    (case-sensitive smallest) spelling of each group, then ordered by
    (length, lowercased name).  When k is given, only the first k names
    are returned.
    """
    if not names:
        return None
    distinct: dict[str, str] = {}
    for name in names:
        key = name.lower()
        if key not in distinct or name < distinct[key]:
            distinct[key] = name
    ordered = sorted(distinct.values(), key=lambda n: (len(n), n.lower(), n))
    return ordered if k is None else ordered[:k]
